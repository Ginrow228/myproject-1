атомарность
транзакции должны выполняться полностью либо вовсе не выполниться

в задании с транзакцией:

begin

insert into books (title, author, published_year, genre) 
values('Приступление и наказание', 'Федор Достоевский', 1866, 'Роман')

insert into borrowed_books (book_id, reader_id, borrow_date, return_date, status)
values(7, 9999, '2025-03-23', null, 'borrowed')

в первом insert'e книга была добавлена
во втором insert'e книга не будет добавлена т.к нет айди читателя 9999, необходимо будет делать откат, и писать заново
то есть если одна из операций в транзакции ломается, то все откатывается, если все данные добавлены/изменены то все будет успешно

согласованность 
данные после каждой трранзакции должны оставаться корректными 

если попытаться добавить в арендованные книги книгу со статусом которого нет - система этого не допустит
insert into borrowed_books (book_id, reader_id, borrow_date, status)
values(1, 2, current_date, 'not released')

то есть система не допускает данные которые некорректны 

изоляция
пока транзакция выполняется, другие не должны видеть ее незавершенные действия
если в одной сессии выполнять тразнакцию
begin;

insert into books (title, author, published_year, genre) 
values('Приступление и наказание', 'Федор Достоевский', 1866, 'Роман')

если в другой сессии посмотреть изменения, то их никто не увидит, кроме сессии в которой собственно происходят действия 

долговечность
после коммита изменения не должны теряться

begin ;
insert into books(title, author, published_year, genre)
values ('new_book', 'new_author', 2000, 'neW_genre')
commit;

если сервер упадет, после перезапуска данные все равно останутся

атомарность - выполняет все или ничего
согласованность - данные всегда корректны
изоляция - разные сессии не будут мешать друг другу
долговечность - закомиченные данные остаются навсегда
